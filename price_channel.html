ทำไมต้องแยกเป็น SAC, CP, BTC อย่างชัดเจน?
1. นโยบายสิทธิ์ความปลอดภัย (Data Isolation)

พนักงานประจำ Cockpit หรือร้านค้าเครือข่าย BTC ไม่ควรเข้าถึงหรือมองเห็นโครงสร้างราคาของ SAC โดยตรง การแยกหน้า/ช่องทางช่วยให้สามารถควบคุมสิทธิ์ตาม $_SESSION['account_type'] ได้อย่างรัดกุม

2. กลุ่มเป้าหมายราคาแตกต่างกัน

• SAC: เน้นราคาขายส่ง, ช่างอู่, และงานโครงการ
• Cockpit: เน้นราคาป้ายหน้าร้าน, บัตรสมาชิก, โปรโมชั่นผ่อน 0%
• BTC: เน้นราคาส่งศูนย์บริการและดีลเลอร์ยาง

3. ประสิทธิภาพการใช้งานของเซลล์

เมื่อเปิดหน้า sac_sale_page.php เซลล์จะเห็นเฉพาะราคาทุกช่องของ SAC (S1-S4) ทันทีใน 1 แถว โดยไม่มีคอลัมน์ของ Cockpit/BTC มาทำให้ตารางกว้างเกินไปและดูยาก

ผลลัพธ์ของการปรับปรุง:

เดิมในฐานข้อมูล ims_product หากสินค้า 1 ตัวมีช่องราคา S1, S2, S3 จะถูกเก็บเป็น 3 แถว ทำให้หน้าเว็บแสดงสินค้าซ้ำกัน 3 บรรทัด
เมื่อปรับปรุงแล้ว: หน้าเว็บจะรวมข้อมูลเป็น 1 บรรทัดต่อ 1 สินค้า และมีคอลัมน์ราคา S1, S2, S3, S4 วางเรียงเคียงกันอย่างสวยงาม

คำสั่ง SQL Pivot Query แยกตามแต่ละช่องทาง
1. SQL View สำหรับช่องทาง SAC (v_product_sac_prices)
CREATE OR REPLACE VIEW v_product_sac_prices AS
SELECT 
    p.product_id,
    p.name_t AS product_name,
    p.brand,
    p.category,
    -- ช่องราคาต่างๆ ของ SAC (ตัวอย่าง S01, S02, S03, S04 หรือ S1, S2, S3, S4)
    MAX(CASE WHEN p.price_code = 'S01' OR p.price_code = 'S1' OR p.price_code = 'S' THEN p.price END) AS price_s1,
    MAX(CASE WHEN p.price_code = 'S02' OR p.price_code = 'S2' THEN p.price END) AS price_s2,
    MAX(CASE WHEN p.price_code = 'S03' OR p.price_code = 'S3' THEN p.price END) AS price_s3,
    MAX(CASE WHEN p.price_code = 'S04' OR p.price_code = 'S4' THEN p.price END) AS price_s4
FROM ims_product p
WHERE p.price_code LIKE 'S%'
GROUP BY p.product_id, p.name_t, p.brand, p.category;
2. SQL View สำหรับช่องทาง Cockpit (v_product_cp_prices)
CREATE OR REPLACE VIEW v_product_cp_prices AS
SELECT 
    p.product_id,
    p.name_t AS product_name,
    p.brand,
    p.category,
    -- ช่องราคาต่างๆ ของ Cockpit (CP1, CP2, CP3, CP4 หรือ C01, C02...)
    MAX(CASE WHEN p.price_code = 'CP01' OR p.price_code = 'C1' OR p.price_code = 'CP' THEN p.price END) AS price_cp1,
    MAX(CASE WHEN p.price_code = 'CP02' OR p.price_code = 'C2' THEN p.price END) AS price_cp2,
    MAX(CASE WHEN p.price_code = 'CP03' OR p.price_code = 'C3' THEN p.price END) AS price_cp3,
    MAX(CASE WHEN p.price_code = 'CP04' OR p.price_code = 'C4' THEN p.price END) AS price_cp4
FROM ims_product p
WHERE p.price_code LIKE 'C%' OR p.price_code LIKE 'CP%'
GROUP BY p.product_id, p.name_t, p.brand, p.category;
3. SQL View สำหรับช่องทาง BTC (v_product_btc_prices)
CREATE OR REPLACE VIEW v_product_btc_prices AS
SELECT 
    p.product_id,
    p.name_t AS product_name,
    p.brand,
    p.category,
    -- ช่องราคาต่างๆ ของ BTC (B1, B2, B3, B4 หรือ BTC01, BTC02...)
    MAX(CASE WHEN p.price_code = 'BTC01' OR p.price_code = 'B1' OR p.price_code = 'BTC' OR p.price_code = 'B' THEN p.price END) AS price_b1,
    MAX(CASE WHEN p.price_code = 'BTC02' OR p.price_code = 'B2' THEN p.price END) AS price_b2,
    MAX(CASE WHEN p.price_code = 'BTC03' OR p.price_code = 'B3' THEN p.price END) AS price_b3,
    MAX(CASE WHEN p.price_code = 'BTC04' OR p.price_code = 'B4' THEN p.price END) AS price_b4
FROM ims_product p
WHERE p.price_code LIKE 'B%' OR p.price_code LIKE 'BTC%'
GROUP BY p.product_id, p.name_t, p.brand, p.category;
การปรับ Model ในโฟลเดอร์ model/

สามารถปรับที่ไฟล์ model/manage_product_process.php, model/manage_product_cockpit_process.php, และ model/manage_product_btc_process.php ให้คืนค่าฟิลด์ราคาแบบหลายช่อง: 

<?php
// ตัวอย่าง: model/manage_product_process.php สำหรับช่องทาง SAC
session_start();
include('../config/connect_db.php');

if ($_POST["action"] === 'GET_PRODUCT_MULTI_CHANNEL') {
    $channel = $_POST['channel'] ?? 'SAC'; // รับค่า SAC, CP, หรือ BTC
    $draw = intval($_POST['draw']);
    $start = intval($_POST['start']);
    $length = intval($_POST['length']);
    $searchValue = trim($_POST['search']['value'] ?? '');

    // เลือก View หรือ Table ตาม Channel
    $targetView = "v_product_sac_prices";
    if ($channel === 'CP')  $targetView = "v_product_cp_prices";
    if ($channel === 'BTC') $targetView = "v_product_btc_prices";

    $searchQuery = " 1=1 ";
    $params = [];
    if (!empty($searchValue)) {
        $searchQuery .= " AND (product_id LIKE :search OR product_name LIKE :search) ";
        $params[':search'] = "%{$searchValue}%";
    }

    // นับจำนวนทั้งหมด
    $countStmt = $conn->prepare("SELECT COUNT(*) FROM {$targetView} WHERE {$searchQuery}");
    $countStmt->execute($params);
    $totalFiltered = $countStmt->fetchColumn();

    // ดึงข้อมูลแบ่งหน้า
    $stmt = $conn->prepare("SELECT * FROM {$targetView} WHERE {$searchQuery} ORDER BY product_id ASC LIMIT :start, :length");
    foreach ($params as $k => $v) $stmt->bindValue($k, $v);
    $stmt->bindValue(':start', $start, PDO::PARAM_INT);
    $stmt->bindValue(':length', $length, PDO::PARAM_INT);
    $stmt->execute();
    $rows = $stmt->fetchAll(PDO::FETCH_ASSOC);

    $data = [];
    $no = $start + 1;
    foreach ($rows as $r) {
        if ($channel === 'SAC') {
            $data[] = [
                "no"           => $no++,
                "product_id"   => $r['product_id'],
                "product_name" => $r['product_name'],
                "brand"        => $r['brand'] ?? '-',
                "price_1"      => $r['price_s1'] ? number_format($r['price_s1'], 2) : '-',
                "price_2"      => $r['price_s2'] ? number_format($r['price_s2'], 2) : '-',
                "price_3"      => $r['price_s3'] ? number_format($r['price_s3'], 2) : '-',
                "price_4"      => $r['price_s4'] ? number_format($r['price_s4'], 2) : '-',
                "actions"      => "<button class='btn btn-info btn-xs btn-detail' data-id='{$r['product_id']}'>Detail</button>"
            ];
        }
        // ... (กรณี CP และ BTC ทำแบบเดียวกัน)
    }

    echo json_encode([
        "draw" => $draw,
        "recordsTotal" => $totalFiltered,
        "recordsFiltered" => $totalFiltered,
        "data" => $data
    ]);
}
?>
